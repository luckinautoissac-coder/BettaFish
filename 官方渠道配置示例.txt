# Flask 服务器配置
HOST=0.0.0.0
PORT=5000

# ===================== Kimi（月之暗面）======================
# 申请地址: https://platform.moonshot.cn/
# 特点: 国内可访问，新用户送 15 元，长上下文
# 成本: ¥0.02/1K tokens

INSIGHT_ENGINE_API_KEY=sk-your-kimi-key
INSIGHT_ENGINE_BASE_URL=https://api.moonshot.cn/v1
INSIGHT_ENGINE_MODEL_NAME=moonshot-v1-128k

# ===================== DeepSeek ======================
# 申请地址: https://platform.deepseek.com/
# 特点: 超低成本，新用户送 500 万 tokens！
# 成本: ¥0.001/1K tokens（输入），¥0.002/1K tokens（输出）

QUERY_ENGINE_API_KEY=sk-your-deepseek-key
QUERY_ENGINE_BASE_URL=https://api.deepseek.com
QUERY_ENGINE_MODEL_NAME=deepseek-chat

MINDSPIDER_API_KEY=sk-your-deepseek-key
MINDSPIDER_BASE_URL=https://api.deepseek.com
MINDSPIDER_MODEL_NAME=deepseek-chat

# ===================== Gemini（需要中转）======================
# 官方地址: https://ai.google.dev/（国内访问困难）
# 中转服务: https://aihubmix.com/?aff=8Ds9
# 特点: 多模态能力强，适合分析图片视频
# 建议: 使用中转服务

MEDIA_ENGINE_API_KEY=sk-your-gemini-or-aihubmix-key
MEDIA_ENGINE_BASE_URL=https://aihubmix.com/v1
MEDIA_ENGINE_MODEL_NAME=gemini-2.0-flash-exp

REPORT_ENGINE_BASE_URL=https://aihubmix.com/v1
